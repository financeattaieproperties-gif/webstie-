        window.onload = function () {
            const countElements = document.querySelectorAll('[id^=count-]');
            const observerOptions = {
                root: null, // use the viewport as the root
                rootMargin: '0px',
                threshold: 0.1 // trigger when at least 10% of the element is visible
            };

            const observer = new IntersectionObserver((entries, observer) => {
                entries.forEach(entry => {
                    if (entry.isIntersecting) {
                        const element = entry.target;
                        const countValue = element.getAttribute('data-count');
                        if (!isNaN(countValue)) {
                            const countUp = new CountUp(element.id, 0, parseFloat(countValue), 0, 2, {
                                useEasing: true,
                                useGrouping: true,
                                separator: ',',
                                decimal: '.',
                                suffix: '+'
                            });

                            if (!countUp.error) {
                                countUp.start(() => {
                                    setTimeout(() => observer.observe(element), 5000);
                                });
                            } else {
                                console.error(countUp.error);
                            }
                        } else {
                            console.error('countValue is not a number:', countValue);
                        }
                    }
                });
            }, observerOptions);

            countElements.forEach(element => observer.observe(element));
        }
    </script>
